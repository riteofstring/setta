
# Base representation
{id, name, value, type}

# Parameter as primitive
{id: uuid, name: x, value: 23, type: int}
result:
x = 23

# Parameter as child section
{id: uuid, name: x, value: childId, type: reference}
result:
x = child.name

# Section with just parameters
{id: uuid, name: x, value: {callable: objName, params: [id1, id2, id3]}, type: obj}
result:
x = {id1.name: id1.value, id2.name: id2.value, id3.name: id3.value}

# Section as a callable with no params
{id: uuid, name: x, value: {callable: objName, params: []}, type: obj}
result:
x = objName()

# Section as a callable with params
{id: uuid, name: x, value: {callable: objName, params: [id1, id2, id3]}, type: obj}
result:
x = objName(**{id1.name: id1.value, id2.name: id2.value, id3.name: id3.value})

# List root
{id: uuid, name: x, value: [id1, id2, id3], type: list}
result:
x = [id1.value, id2.value, id3.value]

# Dict root
{id: uuid, name: x, value: [id1, id2, id3], type: dict}
result:
x = {id1.name: id1.value, id2.name: id2.value, id3.name: id3.value}

# Deck
{id: uuid, name: x, value: [id1, id2, id3], type: deck}
result:
gets spread into whatever the parent container is
